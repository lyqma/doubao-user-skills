## 十二、AutoCAD命令序列参考

```scr
; === 台基层 ===
LAYER NEW BASE COLOR 8 CONTINUOUS
LAYER BASE
_RECTANGLE -8840,-1300 8840,1300
_REGION
_EXTRUDE _L 600
_RECTANGLE -6000,-1000 6000,1000
_REGION
_EXTRUDE _L 400
_RECTANGLE -4000,-800 4000,800
_REGION
_EXTRUDE _L 200

; === 檐柱层 ===
LAYER NEW WOOD_FRAME COLOR 3 CONTINUOUS
LAYER WOOD_FRAME
; 前柱
_RECTANGLE -5420,-1020 -5380,-980
_RECTANGLE -560,-1020 -520,-980
_RECTANGLE 2410,-1020 2450,-980
_RECTANGLE 5380,-1020 5420,-980
; 后柱
_RECTANGLE -5420,980 -5380,1020
_RECTANGLE -560,980 -520,1020
_RECTANGLE 2410,980 2450,1020
_RECTANGLE 5380,980 5420,1020
_REGION
_EXTRUDE _L 6600

; === 斗拱层 ===
LAYER NEW DOUGONG COLOR 6 CONTINUOUS
LAYER DOUGONG
; 栌斗
_RECTANGLE -5600,-1200 -5200,-800
_RECTANGLE -700,-1200 -300,-800
_RECTANGLE 2310,-1200 2550,-800
_RECTANGLE 5200,-1200 5600,-800
_RECTANGLE -5600,800 -5200,1200
_RECTANGLE -700,800 -300,1200
_RECTANGLE 2310,800 2550,1200
_RECTANGLE 5200,800 5600,1200
_REGION
_EXTRUDE _L 264

; === 额枋层 ===
LAYER WOOD_FRAME
; 明间额枋
_RECTANGLE -2340,-1100 -2340+3600,-900
_RECTANGLE -2340,900 -2340+3600,1100
_REGION
_EXTRUDE _L 200

; === 屋檐层 ===
LAYER NEW ROOF COLOR 5 CONTINUOUS
LAYER ROOF
; 檐口板
_RECTANGLE -8640,-1660 8640,1660
_REGION
_EXTRUDE _L 20

; === 宝顶层 ===
LAYER NEW DECORATION COLOR 10 CONTINUOUS
LAYER DECORATION
; 冲天柱
_RECTANGLE -20,16248 20,18558
_RECTANGLE -20,-18558 20,-16248
_REGION
_EXTRUDE _L 2310
; 宝顶球（使用CYLINDER模拟）
_CYLINDER 0,0,16608 240 480
```
